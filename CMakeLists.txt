cmake_minimum_required(VERSION 3.16)
project(ParallelFilesystemInspection LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(inspection scan.cpp)
target_include_directories(inspection PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(inspection PUBLIC cxx_std_17)
target_link_libraries(inspection PUBLIC Threads::Threads)
add_executable(filesystem-inspect main.cpp)
target_link_libraries(filesystem-inspect PRIVATE inspection)
if(MINGW)
    target_link_options(filesystem-inspect PRIVATE -municode)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(scan-test tests/scan_test.cpp)
    target_link_libraries(scan-test PRIVATE inspection)
    add_test(NAME filesystem-inspection COMMAND scan-test)
    set_tests_properties(filesystem-inspection PROPERTIES TIMEOUT 30)
endif()
foreach(target inspection filesystem-inspect scan-test)
    if(TARGET ${target})
        set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /permissive-)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endif()
endforeach()
